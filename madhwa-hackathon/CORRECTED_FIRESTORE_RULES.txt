rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // =========================================================
    // HELPER
    // =========================================================

    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn()
        && exists(
          /databases/$(database)/documents/users/$(request.auth.uid)
        )
        && get(
          /databases/$(database)/documents/users/$(request.auth.uid)
        ).data.role == 'admin';
    }


    // =========================================================
    // USERS
    // =========================================================

    match /users/{userId} {

      // A user can read their own profile.
      // Admin can read all user profiles.
      allow read: if isSignedIn()
                  && (
                    request.auth.uid == userId
                    || isAdmin()
                  );

      // Only admins can create/update/delete user profiles.
      allow create, update, delete: if isAdmin();
    }


    // =========================================================
    // MENUS
    // =========================================================

    match /menus/{date} {

      // Logged-in users can view menus.
      allow read: if isSignedIn();

      // Only admins can create/update/delete menus.
      allow create, update, delete: if isAdmin();
    }


    // =========================================================
    // DAILY STAPLES
    // =========================================================

    match /dailyStaples/{documentId} {

      // Logged-in users can view Daily Staples.
      allow read: if isSignedIn();

      // Only admins can modify Daily Staples.
      allow create, update, delete: if isAdmin();
    }


    // =========================================================
    // SETTINGS
    // =========================================================

    match /settings/{settingId} {

      // Logged-in users can read settings.
      allow read: if isSignedIn();

      // Only admins can modify settings.
      allow create, update, delete: if isAdmin();
    }


    // =========================================================
    // MEAL SELECTIONS
    // =========================================================

    match /mealSelections/{date}/users/{userId} {

      // Employee can read/write their own selection.
      allow read, create, update: if isSignedIn()
                                  && request.auth.uid == userId;

      // Admin can read all employee selections.
      allow read: if isAdmin();
    }


    // =========================================================
    // WORKING MODES
    // =========================================================

    match /workingModes/{date}/users/{userId} {

      // Employee can read/write their own working mode.
      allow read, create, update: if isSignedIn()
                                  && request.auth.uid == userId;

      // Admin can read all working modes.
      allow read: if isAdmin();
    }


    // =========================================================
    // REPORTS
    // =========================================================

    match /reports/{date} {

      // Only admins can access reports.
      allow read, create, update, delete: if isAdmin();
    }


    // =========================================================
    // ANNOUNCEMENTS
    // =========================================================

    match /announcements/{announcementId} {

      // Only admins can create/delete/manage announcements
      // for now.
      allow read, create, update, delete: if isAdmin();
    }


    // =========================================================
    // DENY EVERYTHING ELSE
    // =========================================================

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
